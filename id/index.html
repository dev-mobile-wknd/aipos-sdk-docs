Sekilas pemakaian

val sdk = AIPosSDK.Builder()
    .llmApiKey(BuildConfig.LLM_API_KEY)
    .merchantInfo(merchant)
    .productCatalog(MutableProductCatalog(storeProducts))
    .android(context)
    .build()

// Product suggestions from your app's speech-to-text output
sdk.pushTranscript("looking for a phone for photography, budget 15 million")
sdk.observeSuggestions().collect { suggestions ->
    suggestions.forEach { println("${it.product.name} — ${it.reason}") }
}

// Or a written conversation with the cashier agent
val answer = sdk.sendMessage("find iPhone 15, add 1, what's the total?")
dependencies {
    implementation("com.weekendinc.aipos:aipos-sdk:0.1.0")
}
